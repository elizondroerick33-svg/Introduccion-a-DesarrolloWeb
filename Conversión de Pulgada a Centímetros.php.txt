<?php
// Laboratorio #1 - Conversión de pulgadas a centímetros
// 1 pulgada = 2.54 centímetros

// Se recibe el valor enviado desde el formulario HTML
$pulgadas = $_REQUEST['pulgadas'];

// Se realiza la conversión multiplicando por el factor 2.54
$centimetros = $pulgadas * 2.54;

// Se muestra el resultado en pantalla
echo "Pulgadas: " . $pulgadas . "<br>";
echo "Centímetros: " . $centimetros;
?>
